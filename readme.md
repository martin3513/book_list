## 💻 코드 예시

### Kelly Criterion 구현
```python
def kelly_criterion(win_rate, avg_win, avg_loss):
    """
    Kelly Criterion으로 최적 포지션 크기 계산
    
    Args:
        win_rate: 승률 (0-1)
        avg_win: 평균 수익 (%)
        avg_loss: 평균 손실 (%)
    
    Returns:
        최적 베팅 비율 (0-1)
    """
    kelly = (win_rate * avg_win - (1-win_rate) * avg_loss) / avg_win
    
    # 안전하게 Half Kelly 사용
    return max(0, kelly / 2)

# 예시
win_rate = 0.55
avg_win = 5.0  # 5%
avg_loss = 3.0  # 3%

optimal_size = kelly_criterion(win_rate, avg_win, avg_loss)
print(f"포트폴리오의 {optimal_size*100:.1f}% 투자")
```

### 감정 필터
```python
class EmotionFilter:
    """연속 손실/수익 후 감정 제어"""
    
    def check_state(self, recent_trades):
        """최근 거래로 감정 상태 체크"""
        if len(recent_trades) < 3:
            return "NORMAL"
        
        # 3연속 손실 = 복수 매매 위험
        if all(t.pnl < 0 for t in recent_trades[-3:]):
            return "REVENGE_RISK"
        
        # 5연속 수익 = 과신 위험
        if all(t.pnl > 0 for t in recent_trades[-5:]):
            return "OVERCONFIDENCE_RISK"
        
        return "NORMAL"
```
